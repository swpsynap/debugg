    int* result = (int*)malloc((n + 1) * sizeof(int));
    int count = 0;
    for (int t = 2; t < n - 1; t++)
    {
        int left = 0, right = n - 1, side = 0, ok = 1, valueK = 0;
        for (int step = 1; step <= n; step++)
        {
            if (step > 1 && step != t)
                side = 1 - side;
            int v;
            if (side == 0)
                v = A[left++];
            else
                v = A[right--];
            if (obs[step - 1] != -1 && obs[step - 1] != v)
                ok = 0;
            if (step == k)
                valueK = v;
        }
        if (ok)
        {

            for (int q = 1; q < count; q++)
            {
                if (result[q] == valueK)
                    seen = 1;
            }
            if (!seen)

                count++;
                int q = count;

                {

                    q--;
